# Reproduce the editorial brief

The JSON brief is the editable source; its Markdown view is generated by the Python block below. Run from the repository root. With `BRAND_BRIEF_WRITE=1` it writes the view; without that environment variable it checks the existing view. All other operations are read-only. It validates source paths, copy budgets against the accepted pattern registry, destination references, conditional proof, asset IDs, and immutable candidate inputs against this batch's base. It prints source-bound JSON for the receipt writer.

This is a bounded authoring check, not a production website checker, legal clearance, live product proof or visual review. Existing recipe values, masters and rendered specimens are unchanged. Human assessment is still required for persuasiveness and final composition. The source hash list allows later readers to identify the exact draft checked.

```python
from pathlib import Path
import hashlib, json, os, re, subprocess

root = Path.cwd()
source = 'docs/programs/brand-website/homepage-brief.json'
recipe = 'archive/sessions/2026-09-06-brand-editorial-readiness/VERIFY.md'
brief = json.loads(Path(source).read_text(encoding='utf8'))
patterns = json.loads(Path(brief['density_source']).read_text(encoding='utf8'))
atmosphere_path = 'design/brand-kit/quiet-material/registry.json'
atmosphere = json.loads(Path(atmosphere_path).read_text(encoding='utf8'))
profiles = {p['id']: p for p in patterns['density']['profiles']}
destinations = {d['id'] for d in brief['destinations']}
visuals = {v['id'] for v in brief['illustrations']}
assets = {a['id'] for k in ['assets','gradients'] for a in atmosphere[k]}
assert brief['status'] == 'draft-for-design' and brief['publication_approved'] is False
assert len(destinations) == len(brief['destinations'])
assert len(visuals) == len(brief['illustrations'])
assert len({s['id'] for s in brief['sections']}) == len(brief['sections'])
paths = {source, recipe, atmosphere_path, brief['density_source']}
for key in ['positioning','voice','claims','proof_checkpoint','reference_lock']:
    paths.add(brief[key])
counts = {}
for section in brief['sections']:
    profile = profiles[section['profile']]
    headline_words, body_words = len(section['headline'].split()), len(section['body'].split())
    assert headline_words <= profile['headline_words'], section['id']
    assert body_words <= profile['body_words'], section['id']
    assert section['visual'] in visuals
    assert set(section['asset_ids']) <= assets
    for action in ['primary','supporting']:
        if section[action]: assert section[action]['destination'] in destinations
    for choice in section.get('choices', []):
        assert len(choice['headline'].split()) <= profiles['card']['headline_words']
        assert len(choice['body'].split()) <= profiles['card']['body_words']
        assert choice['destination'] in destinations
        paths.add(choice['source'])
    paths.update(section['proof_sources'])
    counts[section['id']] = {'headline': headline_words, 'body': body_words}
for visual in brief['illustrations']:
    paths.add(visual['source'])
    if 'reuse' in visual: paths.add(visual['reuse'])
    if 'rows' in visual: assert len(visual['rows']) <= profiles[visual['result_profile']]['rows_max']
paths.update(a['evidence'] for a in brief['asset_readiness'])
record = next(s for s in brief['sections'] if s['id'] == 'execution-record')
assert 'Exclude the whole section' in record['publication_gate']
invoice = next(v for v in brief['illustrations'] if v['id'] == 'invoice-fields')
assert invoice['visible_label'] == 'Illustrative example'
claim_scan = r"""
import fs from 'node:fs';
import {loadRegister,compileMatcher} from './scripts/claims-lib.mjs';
const brief=JSON.parse(fs.readFileSync('docs/programs/brand-website/homepage-brief.json','utf8'));
const copy=brief.sections.flatMap(s=>[s.headline,s.body,...['primary','supporting'].map(k=>s[k]?.label||''),...(s.choices||[]).flatMap(c=>[c.headline,c.body])]).join('\n');
const failures=loadRegister(process.cwd()).filter(r=>r.status==='forbidden' && compileMatcher(r).test(copy)).map(r=>r.id);
if(failures.length) throw new Error('Forbidden draft wording: '+failures.join(', '));
console.log(JSON.stringify({forbidden_matches:failures.length}));
"""
claim_result = json.loads(subprocess.check_output(['node','--input-type=module','-e',claim_scan], text=True))
paths.add('scripts/claims-lib.mjs')

lines = ['# Homepage editorial and illustration brief', '',
    '**Draft for design, 6 September 2026. Not approved launch copy.**', '',
    'Generated from [homepage-brief.json](homepage-brief.json); edit that source and run the [reproduction recipe](../../../archive/sessions/2026-09-06-brand-editorial-readiness/VERIFY.md). This is the current homepage authoring brief, not a replacement for positioning, claims, tokens or the accepted specimen story.', '',
    brief['purpose'], '',
    'The retained Quiet Material direction is the reference lock. The adopted positioning supplies the benefit order; the accepted pattern studies supply composition and density rules. Refero’s bundled copywriting guidance supports concrete verbs, optional supporting copy and a clear next action. No new visual direction or page implementation is introduced.', '',
    '## Read the page', '']
for section in brief['sections']:
    suffix = ' — conditional on live proof' if section['id'] == 'execution-record' else ''
    lines += ['### ' + section['headline'] + suffix, '']
    if section['body']: lines += [section['body'], '']
    for choice in section.get('choices', []):
        lines += ['- **' + choice['headline'] + '.** ' + choice['body']]
    if section.get('choices'): lines += ['']
    actions = [section[k]['label'] for k in ['primary','supporting'] if section[k]]
    if actions: lines += ['Actions: ' + ' · '.join(actions), '']
lines += ['## Section briefs', '', '| Section | Reader question | Visual purpose | Detail destination |', '|---|---|---|---|']
for section in brief['sections']:
    visual = next(v for v in brief['illustrations'] if v['id'] == section['visual'])
    dests = [section[k]['destination'] for k in ['primary','supporting'] if section[k]] + [c['destination'] for c in section.get('choices', [])]
    lines += ['| ' + ' | '.join([section['id'], section['reader_question'], visual['job'], ', '.join(dests)]) + ' |']
lines += ['', 'Pacing: ' + ' → '.join(brief['direction']['rhythm']) + '. If retrieval remains unverified, omit the record section and move from shared access to the closing invitation. No blank proof panel remains in the public page.', '',
    'The hero has one demonstration. The discovery section uses three comparable text links, not three button-filled cards. The access diagram is unframed. The optional record is one flat reading surface. The closing Dusk field uses direct light text and a light button without an inner card.', '',
    'All sizes, spacing, typography and surfaces resolve through existing companion tokens. These are authoring budgets, not a claim that a final layout has been tested. Destination IDs below describe required pages; they are not verified URLs.', '',
    '## Illustration briefs', '']
for visual in brief['illustrations']:
    lines += ['### ' + visual['id'].replace('-',' ').capitalize(), '', '**Status: ' + visual['status'].replace('-',' ') + '.** ' + visual['job'], '']
    for key in ['master_plan','visible_label','narrow','alt','motion']:
        if key in visual: lines += ['- ' + key.replace('_',' ').capitalize() + ': ' + visual[key]]
    if 'rows' in visual:
        lines += ['- Display mapping: ' + '; '.join(r['label'] + ' ← ' + (r.get('field') or ' + '.join(r['fields'])) for r in visual['rows']) + '.']
    lines += ['']
lines += ['## Selected asset readiness', '', brief['asset_scope_note'], '', '| Asset | Use | Evidence status | Next step |', '|---|---|---|---|']
for asset in brief['asset_readiness']:
    lines += ['| ' + ' | '.join([asset['id'],asset['use'],asset['status'].replace('-',' '),asset['note']]) + ' |']
lines += ['', 'The new shared-access diagram is the only original explanatory master currently needed. The invoice fixture and current utility icons already exist; the execution-record asset waits for real retrieval. Do not regenerate the atmosphere library or replace the logo to bypass missing history.', '',
    '## Destinations and launch conditions', '']
for dest in brief['destinations']:
    lines += ['- **' + dest['id'] + ':** ' + dest['purpose'] + ' Status: ' + dest['status'] + '.']
lines += ['']
for section in brief['sections']:
    lines += ['- **' + section['id'] + ':** ' + section['publication_gate']]
lines += ['', '## Continue', '',
    'Working authentication is still required for the synthetic invoice execution and own-record retrieval. The current [proof checkpoint](PROOF-QUALIFICATION.md) owns that procedure. The founder has been asked for the creation source of the retained logo and selected opening image. In parallel, this draft can guide the original shared-access illustration and page composition after the remaining kit adoption work; it does not mark T3, T4 or T5 complete.', '',
    'Return to the [completion plan](SYSTEM-COMPLETION.md) or [brand-kit index](../../../design/brand-kit/README.md).', '']
rendered = '\n'.join(lines)
view = Path(brief['human_view'])
if os.environ.get('BRAND_BRIEF_WRITE') == '1': view.write_text(rendered, encoding='utf8', newline='\n')
assert view.read_text(encoding='utf8') == rendered, 'Regenerate the human view'
paths.add(brief['human_view'])

retained = ['design/tokens/active.json']
for name in ['catalogue','foundations','controls','patterns']:
    retained.append('design/tokens/candidates/quiet-material-' + name + '.json')
for folder in ['', '/foundations', '/controls', '/patterns']:
    retained.append('design/brand-kit/quiet-material' + folder + '/registry.json')
for file in retained:
    expected = subprocess.check_output(['git','show',brief['source_baseline'] + ':' + file]).replace(b'\r\n', b'\n')
    assert Path(file).read_text(encoding='utf8').encode('utf8') == expected, file
    paths.add(file)
for file in ['design/brand-kit/README.md','docs/programs/brand-website/PROGRAM.md','docs/programs/brand-website/tracks.yaml','docs/programs/brand-website/SYSTEM-COMPLETION.md','docs/programs/brand-website/system-completion.json','handoff/_general/from-code/2026-09-06-brand-editorial-readiness.md','archive/sessions/2026-09-06-brand-editorial-readiness/access-check.json']:
    paths.add(file)
hashes = {}
for file in sorted(paths):
    path = Path(file)
    assert path.is_file(), file
    data = path.read_bytes() if path.suffix == '.png' else path.read_text(encoding='utf8').encode('utf8')
    hashes[file] = hashlib.sha256(data).hexdigest()
for file in [brief['human_view'], 'design/brand-kit/README.md', 'docs/programs/brand-website/SYSTEM-COMPLETION.md']:
    for dest in re.findall(r'\]\(([^)]+)\)', Path(file).read_text(encoding='utf8')):
        if re.match(r'^[a-z]+:', dest) or dest.startswith('#'): continue
        assert (Path(file).parent / dest.split('#')[0]).exists(), (file, dest)
print(json.dumps({'ok': True, 'sections_checked': len(counts), 'forbidden_copy_matches': claim_result['forbidden_matches'], 'live_output_verified': False,
    'producer': recipe, 'hash_basis': 'UTF-8 LF text; raw PNG', 'word_counts': counts,
    'inputs': hashes, 'limits': 'Draft authoring and unchanged candidate inputs; no live execution, legal clearance, visual or production qualification.'}, indent=2))
```
